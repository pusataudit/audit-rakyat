function doPost(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var data = JSON.parse(e.postData.contents); var reportID = "ADT-" + new Date().getFullYear() + "-" + Math.floor(1000 + Math.random() * 9000); sheet.appendRow([ new Date(), // A data.nama, // B data.entitas, // C data.provinsi, // D data.kota, // E data.kec, // F data.kel, // G data.rw, // H data.whatsapp, // I data.email, // J (BARU) data.kategori, // K data.deskripsi, // L data.linkBukti, // M reportID, // N "OPEN", // O data.kerugian // P ]); return ContentService.createTextOutput(JSON.stringify({"reportID": reportID})) .setMimeType(ContentService.MimeType.JSON); } function doGet(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); if (e.parameter.action == "read") { var data = sheet.getDataRange().getValues(); return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON); } }
